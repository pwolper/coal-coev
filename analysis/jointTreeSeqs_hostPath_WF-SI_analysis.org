# Helper function to finds the simulation directory for the params combination.  
def get_directory(N_host, cV, beta, G):
    baseDir = str(str(here()) + "/slim/trees/jointTreeSeqs")
    dirString = f"/HostPathogen_WF_SI_Nhost{N_host}_cV{cV}_beta{beta}_G{G}"
    d = str(baseDir + dirString)
    if not os.path.isdir(d):
        raise FileNotFoundError(d)
    return d
    
# Function to get the list of unique list of host-pathogen tree Seq files
def get_hostPathFilePairs(N_host, cV, beta, G):
    """ Returns dictionary of host and Pathogen filenames from param dir"""
    params = {"N_host": N_host, "cV": cV, "beta": beta, "G": G}
    d = get_directory(N_host, cV, beta, G)
    
    pairs = []
    for hFile in sorted(glob.glob(f"{d}/*_HostTreeSeq.trees")):
        seed = os.path.basename(hFile).split("_")[0]
        pFile = f"{d}/{seed}_PathogenTreeSeq.trees"
        if not os.path.exists(pFile):
            print(f"PathogenFile not present for seed {seed}.")
            continue
        pairs.append((hFile, pFile)) # list of tuples (host,pathogen) filenames
    print(f"'{d.split('/')[-1]}' contains {len(pairs)} reps")
    return {"params": params, "FilePairs": pairs}

def pedigree_to_nodeID(ts):
    ids = {}
    for s in ts.samples():
        ind = ts.individual(ts.node(s).individual)
        ids[ind.metadata["pedigree_id"]] = int(s)
    assert len(ids) == ts.num_samples
    return ids

from dataclasses import dataclass, replace

@dataclass
class jointTreeSeq:
    hts: tskit.TreeSequence
    pts: tskit.TreeSequence
    infPairs: dict   # infected pairs
    params: dict    # model parameters

    @classmethod
    def load(cls, host_file, patho_file):
        hts, pts = tskit.load(host_file), tskit.load(patho_file)
        meta = pts.metadata['SLiM']['user_metadata']

        infDict = meta['InfDict'][0]
        infPairs = {int(k): int(v[0]) for k, v in infDict.items()}    # add list of integers, for multiple infection later on

        raw = meta['params'][0]
        params = {k: (v[0] if isinstance(v, list) else v) for k, v in raw.items()}
        
        # return cls(hts, pts, params, pairs)
        return cls(hts, pts, infPairs, params)

    def remove_vacants(self):
        """Remove the vacant nodes for haploid treeSeqs"""
        return replace(self,
                       hts=pyslim.remove_vacant(self.hts),
                       pts=pyslim.remove_vacant(self.pts))
        

    def sampled_paired_infected(self, n=None):
        """Sample n pairs of hosts and infecting pathogens"""
        hostIDs, pathogenIDs = pedigree_to_nodeID(self.hts), pedigree_to_nodeID(self.pts)
        pairs = list(self.infPairs.items())
        n = len(pairs) if n is None else n
        if n > len(pairs):
            raise ValueError(f"n={n} exceeds {len(pairs)} infected pairs")
        sampled_pairs = random.sample(pairs, n)
        #print(f"Infected pairs: {len(sampled_pairs)}")
        return replace(self,
                       hts=self.hts.simplify(samples=[hostIDs[h] for p, h in sampled_pairs]),
                       pts=self.pts.simplify(samples=[pathogenIDs[p] for p, h in sampled_pairs]),
                       infPairs=dict(sampled_pairs))

def tmrcas(j):
    """Dictionary with host and pathogen (pairwise) TMRCA"""
    ht, pt = j.hts.first(), j.pts.first()
    host_tmrca = ht.time(ht.root)
    host_mean_t2 = j.hts.diversity(mode="branch") / 2
    pathogen_tmrca = pt.time(pt.root)
    pathogen_mean_t2 = j.pts.diversity(mode="branch") / 2
    
    return {"host_tmrca": host_tmrca,
            "host_mean_t2": host_mean_t2,
            "pathogen_tmrca": pathogen_tmrca,
            "pathogen_mean_t2": pathogen_mean_t2}
