The \texttt{dataclass} /JointTreeSeq/ will hold the host and pathogen tree sequences.
#+begin_src python :session :results output
import tskit, pyslim
from dataclasses import dataclass, field


@dataclass
class jointTreeSeq:
    hts: tskit.TreeSequence
    pts: tskit.TreeSequence
    # params: dict    # model parameters
    infPairs: dict   # infected pairs

    @classmethod
    def load(cls, host_file, patho_file):
        hts, pts = tskit.load(host_file), tskit.load(patho_file)

        infDict = pts.metadata['SLiM']['user_metadata']
        infPairs = {int(k): int(v[0]) for k, v in infDict.items()}    # add list of integers later on

        # parse parameters from metadata as well later
        
        # return cls(hts, pts, params, pairs)
        return cls(hts, pts, infPairs)
#+end_src
